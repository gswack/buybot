Azure commands

az login

Create RG:
$ az group create --name rg-products --location eastus
Show groups:
az group list --output table

Create VNet and subnet:
az network vnet create \
  --resource-group rg-products \
  --name vnet-products \
  --address-prefix 10.0.0.0/16 \
  --subnet-name subnet-public \
  --subnet-prefix 10.0.1.0/24

az network vnet subnet create \
  --resource-group rg-products \
  --vnet-name vnet-products \
  --name subnet-private \
  --address-prefix 10.0.2.0/24

Create public IP for NAT Gateway:
az network public-ip create \
  --resource-group rg-products \
  --name pip-natgw \
  --sku Standard \
  --allocation-method Static

az network nat gateway create \
  --resource-group rg-products \
  --name natgw-products \
  --public-ip-addresses pip-natgw \
  --idle-timeout 10

az network vnet subnet update \
  --resource-group rg-products \
  --vnet-name vnet-products \
  --name subnet-private \
  --nat-gateway natgw-products

Route Table:
az network route-table create \
  --resource-group rg-products \
  --name rt-public

az network vnet subnet update \
  --resource-group rg-products \
  --vnet-name vnet-products \
  --name subnet-public \
  --route-table rt-public


Network Security Group:
az network nsg create \
  --resource-group rg-products \
  --name nsg-public

az network nsg rule create \
  --resource-group rg-products \
  --nsg-name nsg-public \
  --name Allow-SSH \
  --priority 100 \
  --protocol Tcp \
  --destination-port-ranges 22 \
  --access Allow \
  --direction Inbound

az network nsg rule create \
  --resource-group rg-products \
  --nsg-name nsg-public \
  --name Allow-NodeApp \
  --priority 110 \
  --protocol Tcp \
  --destination-port-ranges 3000 \
  --access Allow \
  --direction Inbound

az network vnet subnet update \
  --resource-group rg-products \
  --vnet-name vnet-products \
  --name subnet-public \
  --network-security-group nsg-public

Public IP + VM:
az network public-ip create \
  --resource-group rg-products \
  --name pip-vm1 \
  --sku Standard \
  --allocation-method Static

az vm create \
  --resource-group rg-products \
  --name vm-node1 \
  --vnet-name vnet-products \
  --subnet subnet-public \
  --nsg nsg-public \
  --public-ip-address pip-vm1 \
  --image Ubuntu2204 \
  --admin-username azureuser \
  --generate-ssh-keys \
  --size Standard_B1s

Get the VM's public IP:
az vm show -d --resource-group rg-products --name vm-node1 --query publicIps --output tsv
172.190.14.142
Private IP: 10.0.2.4


SSH:
ssh azureuser@<the-ip-you-got-back>

Install Node.js:
curl -fsSL https://deb.nodesource.com/setup_lts.x | sudo -E bash -
sudo apt-get install -y nodejs
node -v
npm -v

Finally, open app in browser: http://<vm-node1-public-ip>:3000
